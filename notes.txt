______________________two_way_binding_____________________________

--> to manage inputs in form

import React from 'react'
import { useState } from 'react'

const App = () => {

  const [title,setTitle] = useState('')

  const submitHandler=(e)=>{
    e.preventDefault()
    console.log('form submitted')
  }
  return (
    <div>
      <form onSubmit={(e)=>{
        submitHandler(e)
      }}>
        <input type="text" 
        placeholder='Enter ur name' 
        value={title}
        onChange={(e)=>{
          setTitle(e.target.value)
        }}
        />

        <button>submit</button>
      </form>
    </div>
  )
}
export default App
